## 三、“计数3”任务的特殊业务机制与执行闭环

> [!IMPORTANT]
> **用户原话**: *"有的有计数3能重新获取任务列表然后继续执行。每一个能完成的任务都领取了奖励，有的是不行的"*

### 3.1 什么是“计数3”任务？
以 `missionDefId: 44914003`（"在页面内点击3个店铺"）为例，该任务要求用户在指定页面内完成 **3 次**点击行为。

### 3.2 为什么必须“重新获取任务列表”？
1. **进度状态在服务端维护**: 客户端每触发一次行为，服务端累计 `stage.count`（如 `0 -> 1 -> 2 -> 3`）。
2. **密钥动态刷新**: 每次阶段推进后，任务的防重放密钥 `missionXId` 以及领奖状态 `rewardStatus` 都会变更。
3. **不能盲目领奖**: 如果未达到 3 次，调用 `receiveprize` 会返回 `STAGE_COUNT_NOT_ENOUGH` 或 `TASK_NOT_FINISH`。

### 3.3 “计数3”执行的标准自动化算法（伪代码流程）
```kotlin
fun executeCountTask(task: TaskItem) {
    val maxCount = 3 // 从任务标题"点击3个店铺"或 actionConfig 中提取目标计数
    var currentCount = task.currentStageCount // 当前进度 (0~2)
    
    while (currentCount < maxCount) {
        Log.i(TAG, "执行计数任务: ${task.showTitle}, 当前进度: $currentCount/$maxCount")
        
        // 1. 模拟触发点击或访问 actionLink (带上 task.alipayActionLink)
        triggerTaskAction(task)
        Thread.sleep(1500) // 适当停顿防止被风控
        
        // 2. 重新拉取任务列表刷新进度与 missionXId
        val updatedTask = querySingleTask(task.missionDefId) ?: break
        currentCount = updatedTask.currentStageCount
        
        // 3. 检查是否达成完成条件
        if (updatedTask.status == "FINISH" || updatedTask.rewardStatus == "WAITING_RECEIVE") {
            Log.i(TAG, "任务已达成完成条件，立即领取奖励！")
            claimReward(updatedTask)
            break
        }
    }
}
```

## 四、Sesame-TK 集成代码示例

在 Sesame-TK 的架构下，利用现有的 `MtopSender`，可直接编写针对幸运星任务体系的 `LuckyStarRpcCall.kt` 与 `LuckyStarTask.kt`。

### 4.1 RPC 封装层示例 (`LuckyStarRpcCall.kt`)
```kotlin
package fansirsqi.xposed.sesame.task.otherTask2.eleme

import fansirsqi.xposed.sesame.hook.MtopSender
import fansirsqi.xposed.sesame.hook.internal.LocationHelper
import org.json.JSONObject

object LuckyStarRpcCall {
    private const val COLLECTION_ID = "3112"
    private const val BIZ_SCENE = "interact_center"
    private const val SP_PAGE_ID = "26051124184460076"

    /**
     * 查询全量任务列表
     */
    fun queryTaskList(): MtopSender.Result {
        val data = JSONObject().apply {
            put("missionCollectionId", COLLECTION_ID)
            put("bizScene", BIZ_SCENE)
            put("launchChannels", "[]")
            put("accountPlan", "HAVANA_COMMON")
            put("spPageId", SP_PAGE_ID)
            put("instance", "INNER")
            LocationHelper.getLocation()?.let {
                put("latitude", it.optString("latitude", "25.274027"))
                put("longitude", it.optString("longitude", "110.299078"))
                put("locationInfos", "[{\"lng\":\"${it.optString("longitude")}\",\"lat\":\"${it.optString("latitude")}\"}]")
            }
        }
        return MtopSender.send("mtop.ele.biz.growth.task.core.querytask", "1.0", data.toString(), false, null)
    }

    /**
     * 刷新指定任务状态
     */
    fun querySingleTask(missionId: Long): MtopSender.Result {
        val data = JSONObject().apply {
            put("missionCollectionId", COLLECTION_ID)
            put("bizScene", BIZ_SCENE)
            put("launchChannels", "[]")
            put("missionIds", "[$missionId]")
            put("accountPlan", "HAVANA_COMMON")
            put("spPageId", SP_PAGE_ID)
            put("instance", "INNER")
        }
        return MtopSender.send("mtop.ele.biz.growth.task.core.querytask", "1.0", data.toString(), false, null)
    }

    /**
     * 领取任务奖励
     */
    fun receivePrize(missionId: Long, missionXId: String): MtopSender.Result {
        val data = JSONObject().apply {
            put("missionCollectionId", COLLECTION_ID.toInt())
            put("missionId", missionId)
            put("missionXId", missionXId)
            put("extInfo", "{\"triggerScene\":\"MISSION_LIST\"}")
            put("count", 1)
            put("bizScene", BIZ_SCENE)
            put("accountPlan", "HAVANA_COMMON")
            put("spPageId", SP_PAGE_ID)
            put("instance", "INNER")
        }
        return MtopSender.send("mtop.ele.biz.growth.task.core.receiveprize", "1.0", data.toString(), false, null)
    }

    /**
     * 刷新幸运星资产余额
     */
    fun queryStarBalance(): MtopSender.Result {
        val data = JSONObject().apply {
            put("bizScene", BIZ_SCENE)
            put("version", "1.4.22")
            put("modules", "[\"property\"]")
            put("channel", "ES0021178012")
            put("spPageId", SP_PAGE_ID)
            put("instance", "INNER")
        }
        return MtopSender.send("mtop.alsc.interact.et.interact.center.homepage", "1.4.22", data.toString(), false, null)
    }
}
```

### 4.2 业务调度器核心逻辑示例 (`LuckyStarTask.kt`)
```kotlin
package fansirsqi.xposed.sesame.task.otherTask2.eleme

import fansirsqi.xposed.sesame.util.Log
import kotlinx.coroutines.delay
import org.json.JSONObject

object LuckyStarTask {
    private const val TAG = "🌟饿了么幸运星"

    private val BLACKLIST_KEYWORDS = listOf("下单", "购买", "支付", "实付", "开通", "会员", "SVIP")

    suspend fun execute() {
        Log.other(TAG, "开始执行幸运星任务流...")
        
        // 1. 查询当前幸运星余额
        val propRes = LuckyStarRpcCall.queryStarBalance()
        if (propRes.ok) {
            val starAmount = propRes.body()?.optJSONObject("property")?.optJSONObject("data")?.optJSONObject("STAR")?.optInt("amount", -1)
            Log.other(TAG, "当前幸运星余额: $starAmount")
        }

        // 2. 拉取全量任务列表
        val taskRes = LuckyStarRpcCall.queryTaskList()
        if (!taskRes.ok) {
            Log.error(TAG, "拉取任务列表失败: ${taskRes.failReason}")
            return
        }

        val mlist = taskRes.body()?.optJSONArray("mlist") ?: return
        Log.other(TAG, "获取到 ${mlist.length()} 个任务")

        for (i in 0 until mlist.length()) {
            val taskObj = mlist.optJSONObject(i) ?: continue
            val missionDefId = taskObj.optLong("missionDefId", -1L)
            val showTitle = taskObj.optString("showTitle", taskObj.optString("name", "未知任务"))
            val status = taskObj.optString("status")
            val rewardStatus = taskObj.optString("rewardStatus")
            val receiveStatus = taskObj.optString("receiveStatus")
            val missionXId = taskObj.optString("missionXId")

            // 黑名单过滤
            if (BLACKLIST_KEYWORDS.any { showTitle.contains(it) }) {
                Log.other(TAG, "[$showTitle] 命中黑名单关键词，跳过")
                continue
            }

            // 已领奖跳过
            if (receiveStatus == "HAVERECEIVED") {
                Log.other(TAG, "[$showTitle] 已完成并领奖，跳过")
                continue
            }

            // 如果已经达到可领奖状态，直接领奖
            if (status == "FINISH" || rewardStatus == "WAITING_RECEIVE" || rewardStatus == "SUCCESS") {
                claim(missionDefId, missionXId, showTitle)
                continue
            }

            // 处理进行中任务
            val actionConfig = taskObj.optJSONObject("actionConfig")
            val actionType = actionConfig?.optString("actionType") ?: "PAGEVIEW"
            val stageTime = actionConfig?.optJSONObject("actionValue")?.optInt("pageStageTime", 15) ?: 15

            Log.other(TAG, "[$showTitle] 类型: $actionType, 开始执行...")

            if (showTitle.contains("点击3个店铺") || actionType == "THIRD") {
                // 计数类循环处理
                handleCountTask(missionDefId, showTitle)
            } else {
                // 浏览类等待
                delay(stageTime * 1000L)
                // 刷新单任务状态获取最新的 missionXId
                val refreshRes = LuckyStarRpcCall.querySingleTask(missionDefId)
                val refreshedTask = refreshRes.body()?.optJSONArray("mlist")?.optJSONObject(0)
                val newXId = refreshedTask?.optString("missionXId", missionXId) ?: missionXId
                claim(missionDefId, newXId, showTitle)
            }
        }

        // 3. 最终再次查询资产确认收益
        val finalProp = LuckyStarRpcCall.queryStarBalance()
        val finalStar = finalProp.body()?.optJSONObject("property")?.optJSONObject("data")?.optJSONObject("STAR")?.optInt("amount", -1)
        Log.other(TAG, "任务执行完毕，最新幸运星余额: $finalStar")
    }

    private suspend fun handleCountTask(missionId: Long, title: String) {
        for (step in 1..3) {
            Log.other(TAG, "[$title] 执行第 $step/3 次阶段动作...")
            delay(1500L)
            // 每次重新获取任务列表以推进服务端进度和更新 missionXId
            val updateRes = LuckyStarRpcCall.querySingleTask(missionId)
            val updated = updateRes.body()?.optJSONArray("mlist")?.optJSONObject(0) ?: break
            if (updated.optString("status") == "FINISH" || updated.optString("rewardStatus") == "WAITING_RECEIVE") {
                val newXId = updated.optString("missionXId")
                claim(missionId, newXId, title)
                break
            }
        }
    }

    private fun claim(missionId: Long, xId: String, title: String) {
        val claimRes = LuckyStarRpcCall.receivePrize(missionId, xId)
        if (claimRes.ok) {
            Log.other(TAG, "🎉 [$title] 奖励领取成功！")
        } else {
            Log.error(TAG, "❌ [$title] 领奖失败: ${claimRes.retMsg}")
        }
    }
}
```
